cmake_minimum_required(VERSION 3.21)

project(XVisionSdkAdapterFixture LANGUAGES NONE)

if(NOT DEFINED XVISION_SOURCE_DIR OR XVISION_SOURCE_DIR STREQUAL "")
    message(FATAL_ERROR "XVISION_SOURCE_DIR is required")
endif()

option(XVISION_ENABLE_OPENCV "Exercise the XVision OpenCV adapter" OFF)
option(XVISION_ENABLE_ONNXRUNTIME "Exercise the XVision ONNX Runtime adapter" OFF)

include("${XVISION_SOURCE_DIR}/3rdparty/OpenCV.cmake")
include("${XVISION_SOURCE_DIR}/3rdparty/OnnxRuntime.cmake")

if(XVISION_ENABLE_OPENCV)
    if(NOT TARGET XVision::OpenCV)
        message(FATAL_ERROR "XVision::OpenCV was not created")
    endif()
    get_target_property(_opencv_include XVision::OpenCV INTERFACE_INCLUDE_DIRECTORIES)
    if(NOT _opencv_include OR NOT EXISTS "${_opencv_include}/opencv2/core.hpp"
            OR NOT EXISTS "${_opencv_include}/opencv2/imgproc.hpp"
            OR NOT EXISTS "${_opencv_include}/opencv2/features2d.hpp"
            OR NOT EXISTS "${_opencv_include}/opencv2/xfeatures2d.hpp"
            OR NOT EXISTS "${_opencv_include}/opencv2/dnn_superres.hpp"
            OR NOT EXISTS "${_opencv_include}/opencv2/video/background_segm.hpp")
        message(FATAL_ERROR "XVision::OpenCV has an invalid include directory")
    endif()
    if(WIN32)
        get_target_property(_opencv_type XVision::OpenCV TYPE)
        if(NOT _opencv_type STREQUAL "SHARED_LIBRARY")
            message(FATAL_ERROR "Windows OpenCV must link its import library as a SHARED target")
        endif()
        get_target_property(_opencv_library XVision::OpenCV IMPORTED_IMPLIB)
    else()
        get_target_property(_opencv_library XVision::OpenCV IMPORTED_LOCATION)
    endif()
    if(NOT _opencv_library OR NOT EXISTS "${_opencv_library}")
        message(FATAL_ERROR "XVision::OpenCV has an invalid imported library")
    endif()
    if(EXPECT_VIDEO_RUNTIME_FILES)
        set(_video_runtime "${XVISION_OPENCV_ROOT}/bin/opencv_videoio_ffmpeg4100_64.dll")
        if(NOT _video_runtime IN_LIST XVISION_OPENCV_RUNTIME_DLLS)
            message(FATAL_ERROR "OpenCV video runtime is missing from deployment DLLs")
        endif()
    endif()
elseif(TARGET XVision::OpenCV)
    message(FATAL_ERROR "Disabled OpenCV adapter unexpectedly created a target")
endif()

if(XVISION_ENABLE_ONNXRUNTIME)
    if(NOT TARGET XVision::ONNXRuntime)
        message(FATAL_ERROR "XVision::ONNXRuntime was not created")
    endif()
    get_target_property(_onnx_include XVision::ONNXRuntime INTERFACE_INCLUDE_DIRECTORIES)
    if(NOT _onnx_include OR NOT EXISTS "${_onnx_include}/onnxruntime_cxx_api.h"
            OR NOT EXISTS "${_onnx_include}/onnxruntime_c_api.h")
        message(FATAL_ERROR "XVision::ONNXRuntime has an invalid include directory")
    endif()
    if(WIN32)
        get_target_property(_onnx_type XVision::ONNXRuntime TYPE)
        if(NOT _onnx_type STREQUAL "SHARED_LIBRARY")
            message(FATAL_ERROR "Windows ONNX Runtime must link its import library as a SHARED target")
        endif()
        get_target_property(_onnx_library XVision::ONNXRuntime IMPORTED_IMPLIB)
    else()
        get_target_property(_onnx_library XVision::ONNXRuntime IMPORTED_LOCATION)
    endif()
    if(NOT _onnx_library OR NOT EXISTS "${_onnx_library}")
        message(FATAL_ERROR "XVision::ONNXRuntime has an invalid imported library")
    endif()
elseif(TARGET XVision::ONNXRuntime)
    message(FATAL_ERROR "Disabled ONNX Runtime adapter unexpectedly created a target")
endif()
